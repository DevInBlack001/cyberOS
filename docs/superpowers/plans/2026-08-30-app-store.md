# CyberOS App Store and Plugin Market Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A department app store where every published package was built in a clean chroot with no network, linted, policy-checked, malware-scanned, and approved by two maintainers before a student can install it.

**Architecture:** A submission is a pull request adding `apps/<pkgname>/` with a PKGBUILD, an AppStream metainfo file, and a rationale. `tools/verify-submission.sh` runs nine automated gates; a PR failing any of them is not reviewed by a human. Approved packages are signed and published to `[cyberos-apps]`, promoting from `testing` to `stable` on the channel cadence. `cyberos-store` is a thin browser over that repository — it reads an AppStream catalogue and shells out to pacman, which does the download and signature verification.

**Tech Stack:** Bash, `namcap` 3.6, `devtools` 1.5 (`extra-x86_64-build`), `appstream` 1.1.6 (`appstreamcli`), `clamav` 1.5, pacman `repo-add`, GPG, Python 3 + `python-gobject`/`gtk4`/`libadwaita`, `bats` 1.14 and `pytest` for tests, GitHub Actions.

**Spec:** `docs/SPEC.md` §5 (app store and plugin market), §4.6 (promotion follows the channels)

Create `tools/lib/gates.sh`:

```bash
#!/usr/bin/env bash
# CyberOS submission gates.
#
# Each gate takes a submission directory, returns 0 to pass, and prints why it
# failed. A submission that fails ANY gate is not reviewed by a human -- the
# point is that reviewer attention is spent on judgement, not on catching things
# a script catches.

GATE_FAILURES=()

_fail() { printf '%s\n' "$*"; GATE_FAILURES+=("$*"); return 1; }

# G1 -- lint the PKGBUILD itself.
gate_pkgbuild_lint() {
  local dir=$1
  [[ -f $dir/PKGBUILD ]] || _fail "G1: no PKGBUILD in $dir" || return 1
  command -v namcap >/dev/null || _fail "G1: namcap is not installed" || return 1
  local out
  out=$(namcap "$dir/PKGBUILD" 2>&1) || true
  if grep -qE '\(E\)|ERROR' <<<"$out"; then
    _fail "G1: namcap errors on PKGBUILD:"$'\n'"$out"
    return 1
  fi
  return 0
}

# G2 -- every source must be authenticated: https with a checksum, or a git URL
# pinned to a commit. This is the gate that stops an upstream tarball being
# swapped out under us.
gate_sources_pinned() {
  local dir=$1 srcline sumline
  [[ -f $dir/PKGBUILD ]] || _fail "G2: no PKGBUILD in $dir" || return 1
  srcline=$(grep -E '^[[:space:]]*source=' "$dir/PKGBUILD" || true)
  [[ -n $srcline ]] || return 0    # no sources is fine (meta packages)

  if grep -qE "(^|['\"( ])http://" <<<"$srcline"; then
    _fail "G2: source uses plain http; use https"
    return 1
  fi

  sumline=$(grep -E '^[[:space:]]*(sha256sums|b2sums)=' "$dir/PKGBUILD" || true)
  if grep -qE 'git\+' <<<"$srcline"; then
    if ! grep -qE '#(commit|tag)=' <<<"$srcline"; then
      _fail "G2: git source is not pinned to a commit or tag"
      return 1
    fi
    return 0
  fi

  [[ -n $sumline ]] || _fail "G2: no sha256sums or b2sums" || return 1
  if grep -q "SKIP" <<<"$sumline"; then
    _fail "G2: SKIP checksum on a non-VCS source"
    return 1
  fi
  return 0
}

# G9 -- a package with no licence cannot be redistributed by the department.
gate_licence() {
  local dir=$1 line
  line=$(grep -E '^[[:space:]]*license=' "$dir/PKGBUILD" || true)
  [[ -n $line ]] || _fail "G9: no license= in PKGBUILD" || return 1
  if grep -qiE "\('?(unknown|custom)'?\)" <<<"$line"; then
    _fail "G9: license is 'unknown' or bare 'custom'; name it, or use custom:<name> and ship the text"
    return 1
  fi
  return 0
}

# G8 -- the catalogue entry the store will render.
gate_metainfo() {
  local dir=$1 file
  file=$(find "$dir" -maxdepth 1 -name '*.metainfo.xml' | head -1)
  [[ -n $file ]] || _fail "G8: no *.metainfo.xml in $dir" || return 1
  command -v appstreamcli >/dev/null || _fail "G8: appstreamcli is not installed" || return 1
  local out
  if ! out=$(appstreamcli validate --explain "$file" 2>&1); then
    _fail "G8: appstreamcli rejected the metainfo:"$'\n'"$out"
    return 1
  fi
  return 0
}
```

Create `tools/verify-submission.sh`:

```bash
#!/usr/bin/env bash
# verify-submission.sh <submission-dir>
#
# Run every automated gate over a submission. Used by CI and runnable locally
# before opening the pull request.
set -uo pipefail

HERE=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
# shellcheck source=lib/gates.sh
source "$HERE/lib/gates.sh"

DIR=${1:-}
[[ -d $DIR ]] || { printf 'usage: verify-submission.sh <submission-dir>\n' >&2; exit 2; }

GATES=(gate_pkgbuild_lint gate_sources_pinned gate_licence gate_metainfo)

failed=0
for g in "${GATES[@]}"; do
  printf '=== %s\n' "$g"
  if "$g" "$DIR"; then
    printf '    pass\n'
  else
    printf '    FAIL\n'
    failed=1
  fi
done

if (( failed )); then
  printf '\n%s gate(s) failed. A submission that fails a gate is not reviewed.\n' "${#GATE_FAILURES[@]}"
  exit 1
fi
printf '\nAll automated gates passed. Ready for two maintainer reviews.\n'
```

```bash
# build_uses_network <dir> -- exit 0 if build() or package() reference a network
# tool. Curl-to-shell at build time is the classic supply-chain vector, and a
# reviewer reading a 400-line PKGBUILD will miss it.
build_uses_network() {
  local dir=$1 body tool
  # Extract only the bodies of build() and package(), so a comment or a source=
  # URL elsewhere does not trip the check.
  body=$(awk '
    /^[[:space:]]*(build|package|prepare)[[:space:]]*\(\)[[:space:]]*\{/ { depth=1; next }
    depth > 0 {
      depth += gsub(/\{/, "{") - gsub(/\}/, "}")
      if (depth <= 0) { depth = 0; next }
      sub(/#.*/, "")
      print
    }
  ' "$dir/PKGBUILD")

  for tool in curl wget git svn hg pip npm cargo\ fetch go\ get; do
    if grep -qE "(^|[;&|[:space:]])${tool}([[:space:]]|$)" <<<"$body"; then
      printf 'G4: build functions call %s -- the build must not touch the network\n' "$tool"
      return 0
    fi
  done
  return 1
}

# G3 + G4 -- build in a clean chroot with networking off after sources are
# fetched. If it does not build here, it does not build on a student machine.
gate_clean_build() {
  local dir=$1 outdir=${2:-$dir/out}
  if build_uses_network "$dir"; then
    _fail "G4: the build reaches the network"
    return 1
  fi
  command -v extra-x86_64-build >/dev/null \
    || _fail "G3: devtools is not installed (need extra-x86_64-build)" || return 1

  mkdir -p "$outdir"
  local log="$outdir/build.log"
  if ! ( cd "$dir" && PKGDEST="$(cd "$outdir" && pwd)" \
           extra-x86_64-build -- -- --nocheck ) >"$log" 2>&1; then
    _fail "G3: clean-chroot build failed, see $log"$'\n'"$(tail -30 "$log")"
    return 1
  fi
  return 0
}

# G5 -- lint the built package, which catches what linting the recipe cannot:
# missing runtime deps, world-writable files, insecure RPATH.
gate_package_lint() {
  local pkgfile=$1 out
  [[ -f $pkgfile ]] || _fail "G5: no package file at $pkgfile" || return 1
  out=$(namcap "$pkgfile" 2>&1) || true
  if grep -qE '\(E\)|ERROR' <<<"$out"; then
    _fail "G5: namcap errors on the built package:"$'\n'"$out"
    return 1
  fi
  return 0
}
```

```bash
# G6 -- what the package puts on a student's disk.
#
# setuid is refused outright: no submission needs it, and a setuid binary in a
# reviewed repository is a privilege-escalation primitive handed to every
# student. Paths under sudoers.d, polkit rules and systemd units need the pull
# request to be explicitly flagged for privilege review.
gate_filesystem_policy() {
  local pkgfile=$1 allow_priv=0
  [[ ${2:-} == --allow-privileged ]] && allow_priv=1
  [[ -f $pkgfile ]] || _fail "G6: no package file at $pkgfile" || return 1

  local listing rc=0
  listing=$(bsdtar -tvf "$pkgfile") || _fail "G6: cannot read $pkgfile" || return 1

  # setuid / setgid, from the mode column of the listing.
  if grep -qE '^[-d][rwx-]{2}[sS][rwx-]{3}|^[-d][rwx-]{5}[sS]' <<<"$listing"; then
    _fail "G6: package contains a setuid or setgid file"
    rc=1
  fi

  local path
  while read -r path; do
    case $path in
      .PKGINFO|.BUILDINFO|.MTREE|.INSTALL|./) continue ;;
      usr/*|opt/*|etc/*) ;;
      *) _fail "G6: file outside /usr, /opt and /etc: $path"; rc=1; continue ;;
    esac
    case $path in
      etc/sudoers.d/*|etc/polkit-1/rules.d/*|usr/lib/systemd/system/*|etc/systemd/system/*)
        if (( allow_priv )); then
          printf 'G6: privileged path accepted under review: %s\n' "$path"
        else
          _fail "G6: privileged path needs an explicit privilege review: $path"
          rc=1
        fi ;;
    esac
  done < <(bsdtar -tf "$pkgfile" | sed 's@^\./@@' | grep -v '/$')

  return $rc
}

# G7 -- scan the payload. It will not catch a targeted attack, but it catches a
# submitter who packaged something they downloaded and did not read.
gate_malware_scan() {
  local pkgfile=$1 tmp out
  command -v clamscan >/dev/null || _fail "G7: clamav is not installed" || return 1
  tmp=$(mktemp -d)
  bsdtar -xf "$pkgfile" -C "$tmp" 2>/dev/null || true
  if ! out=$(clamscan -ri --no-summary "$tmp" 2>&1); then
    rm -rf "$tmp"
    _fail "G7: clamav flagged the payload:"$'\n'"$out"
    return 1
  fi
  rm -rf "$tmp"
  return 0
}
```

```bash
#!/usr/bin/env bash
# publish-apps.sh --channel <testing|stable> --key <KEYID>
#
# Sign approved packages and publish them to [cyberos-apps] for one channel,
# then regenerate the AppStream catalogue the store reads.
#
# Packages land in testing first and promote to stable on the monthly cadence.
# Publishing straight to stable is for a security fix, and needs the same CVE
# citation that tools/security-backport.sh requires.
set -euo pipefail

die() { printf 'publish-apps: %s\n' "$*" >&2; exit 1; }

CHANNEL= KEY= IN=dist/approved OUT=dist/cyberos-repo UNSIGNED=0 DRYRUN=0
while [[ $# -gt 0 ]]; do
  case $1 in
    --channel)  CHANNEL=$2; shift 2 ;;
    --key)      KEY=$2; shift 2 ;;
    --in)       IN=$2; shift 2 ;;
    --out)      OUT=$2; shift 2 ;;
    --unsigned) UNSIGNED=1; shift ;;
    --dry-run)  DRYRUN=1; shift ;;
    *) die "unknown argument: $1" ;;
  esac
done

case $CHANNEL in
  testing|stable) ;;
  *) die "--channel must be testing or stable (got '${CHANNEL:-nothing}')" ;;
esac
[[ -n $KEY || $UNSIGNED -eq 1 ]] \
  || die "signing key required: pass --key <KEYID>, or --unsigned for a local test repo"

shopt -s nullglob
pkgs=("$IN"/*.pkg.tar.zst)
(( ${#pkgs[@]} )) || die "no packages in $IN"

dest="$OUT/$CHANNEL/x86_64"
printf 'publishing %s package(s) to %s\n' "${#pkgs[@]}" "$dest"
[[ $DRYRUN -eq 1 ]] && exit 0

mkdir -p "$dest"
for p in "${pkgs[@]}"; do
  cp -- "$p" "$dest/"
  [[ $UNSIGNED -eq 0 ]] && gpg --detach-sign --local-user "$KEY" --yes "$dest/$(basename "$p")"
done

if [[ $UNSIGNED -eq 0 ]]; then
  repo-add --sign --key "$KEY" "$dest/cyberos-apps.db.tar.gz" "$dest"/*.pkg.tar.zst
else
  repo-add "$dest/cyberos-apps.db.tar.gz" "$dest"/*.pkg.tar.zst
fi

# The catalogue the store renders. Generated from the metainfo files inside the
# packages, so it cannot describe something that was not published.
catalogue="$OUT/$CHANNEL/catalogue"
mkdir -p "$catalogue"
tmp=$(mktemp -d); trap 'rm -rf "$tmp"' EXIT
for p in "$dest"/*.pkg.tar.zst; do
  bsdtar -xf "$p" -C "$tmp" usr/share/metainfo 2>/dev/null || true
done
if [[ -d $tmp/usr/share/metainfo ]]; then
  appstreamcli compose --origin cyberos-apps --result-root "$catalogue" \
    --data-dir "$catalogue" --prefix /usr "$tmp"
else
  printf 'warning: no metainfo files found; the catalogue will be empty\n' >&2
fi

printf 'published to %s\n' "$dest"
```

```python
#!/usr/bin/env python3
"""cyberos-store -- browse and install department-verified packages.

This is a browser over a pacman repository. It does not download, unpack or
install anything itself: pacman does that, and pacman checks the signatures.
Every package it can offer was built in a clean chroot, linted, policy-checked
and approved by two maintainers before it was published.
"""
import os
import re
import subprocess
import sys

REPO = "cyberos-apps"
PLUGIN_PREFIX = "cyberos-plugin-"

# pacman package names: letters, digits, @ . _ + -, not starting with a dash.
PKGNAME = re.compile(r"^[a-zA-Z0-9@._+][a-zA-Z0-9@._+-]*$")

_SL_ROW = re.compile(r"^(\S+)\s+(\S+)\s+(\S+)(\s+\[installed[^\]]*\])?\s*$")


def parse_pacman_sl(text):
    """Parse `pacman -Sl <repo>` output into rows."""
    rows = []
    for line in text.splitlines():
        m = _SL_ROW.match(line.strip())
        if not m:
            continue
        repo, name, version, installed = m.groups()
        rows.append({"repo": repo, "name": name, "version": version,
                     "installed": installed is not None})
    return rows


def classify(name, component_type):
    """An app or a plugin. The AppStream type wins; the name is the fallback."""
    if component_type == "addon":
        return "plugin"
    if component_type is not None:
        return "app"
    return "plugin" if name.startswith(PLUGIN_PREFIX) else "app"


def install_argv(name):
    """The exact command used to install. pacman does the work, not us."""
    if not PKGNAME.match(name):
        raise ValueError(f"not a package name: {name!r}")
    return ["pkexec", "pacman", "-S", "--needed", "--noconfirm", name]


def listing():
    out = subprocess.run(["pacman", "-Sl", REPO], capture_output=True,
                         text=True, check=False)
    if out.returncode != 0:
        sys.exit(f"cyberos-store: cannot read [{REPO}]: {out.stderr.strip()}\n"
                 f"Is the department repository configured in /etc/pacman.conf?")
    return parse_pacman_sl(out.stdout)


def cmd_list(rows):
    for r in sorted(rows, key=lambda r: r["name"]):
        kind = classify(r["name"], None)
        mark = "*" if r["installed"] else " "
        print(f"{mark} {kind:<7} {r['name']:<32} {r['version']}")
    print("\n* = installed")


def cmd_search(rows, term):
    term = term.lower()
    cmd_list([r for r in rows if term in r["name"].lower()])


def cmd_install(name):
    argv = install_argv(name)
    print(" ".join(argv))
    raise SystemExit(subprocess.run(argv, check=False).returncode)


def cmd_gui(rows):
    import gi
    gi.require_version("Gtk", "4.0")
    gi.require_version("Adw", "1")
    from gi.repository import Gtk, Adw, Gio  # noqa: E402

    # Follow the desktop's light/dark setting, the same way the installer does.
    mode = "dark"
    try:
        with open(os.path.expanduser("~/.config/cyberos/mode")) as fh:
            mode = fh.read().strip() or "dark"
    except OSError:
        pass

    class Store(Adw.Application):
        def do_activate(self):
            Adw.StyleManager.get_default().set_color_scheme(
                Adw.ColorScheme.FORCE_DARK if mode == "dark" else Adw.ColorScheme.FORCE_LIGHT)
            win = Adw.ApplicationWindow(application=self, title="CyberOS Store",
                                        default_width=880, default_height=620)
            box = Gtk.Box(orientation=Gtk.Orientation.VERTICAL)
            box.append(Adw.HeaderBar())

            listbox = Gtk.ListBox(selection_mode=Gtk.SelectionMode.NONE)
            listbox.add_css_class("boxed-list")
            for r in sorted(rows, key=lambda r: r["name"]):
                row = Adw.ActionRow(title=r["name"],
                                    subtitle=f"{classify(r['name'], None)} · {r['version']}")
                if r["installed"]:
                    row.add_suffix(Gtk.Label(label="installed"))
                else:
                    btn = Gtk.Button(label="Install", valign=Gtk.Align.CENTER)
                    btn.add_css_class("suggested-action")
                    btn.connect("clicked", self.on_install, r["name"], row)
                    row.add_suffix(btn)
                listbox.append(row)

            clamp = Adw.Clamp(maximum_size=760, margin_top=18, margin_bottom=18,
                              margin_start=18, margin_end=18, child=listbox)
            scroll = Gtk.ScrolledWindow(vexpand=True, child=clamp)
            box.append(scroll)
            win.set_content(box)
            win.present()

        def on_install(self, button, name, row):
            button.set_sensitive(False)
            button.set_label("Installing…")
            proc = Gio.Subprocess.new(install_argv(name), Gio.SubprocessFlags.NONE)
            proc.wait_async(None, self.on_done, (button, row))

        def on_done(self, proc, result, data):
            button, row = data
            ok = proc.get_successful()
            button.set_label("Installed" if ok else "Failed")
            if ok:
                button.add_css_class("dim-label")
            else:
                button.add_css_class("destructive-action")
                button.set_sensitive(True)

    Store(application_id="dev.cyberos.Store").run([])


def main():
    args = sys.argv[1:]
    if args and args[0] == "--install":
        if len(args) < 2:
            sys.exit("cyberos-store: --install needs a package name")
        return cmd_install(args[1])

    rows = listing()
    if not args or args[0] == "--gui":
        return cmd_gui(rows)
    if args[0] == "--list":
        return cmd_list(rows)
    if args[0] == "--search":
        if len(args) < 2:
            sys.exit("cyberos-store: --search needs a term")
        return cmd_search(rows, args[1])
    sys.exit("usage: cyberos-store [--gui | --list | --search TERM | --install NAME]")


if __name__ == "__main__":
    main()
```

  matches the running command line, heredocs included, and kills the shell running it.
